# Pinned pip version with hash verification.
#
# pip is bootstrapped from whatever version the venv ships with — which is
# necessarily unverified (it is what processes --require-hashes, so it cannot
# verify itself on first use). This manifest pins and hash-verifies the
# UPGRADE, ensuring the pip version used for all subsequent installs is known
# and verified. The venv-bootstrap pip is accepted as a platform-provided
# trust root, same as the Python interpreter itself.
#
# To update: change the version, download the new wheel, and replace the hash:
#   pip download pip==<new-version> --no-deps -d /tmp/pip-dl
#   pip hash /tmp/pip-dl/pip-<new-version>-py3-none-any.whl

pip==26.2.2 \
    --hash=sha256:fd7f4dfa8190e014013b9f08f3e973f4e4c26ffd179a204895533a7254813a70

# setuptools is upgraded so the editable install (--no-build-isolation) can
# use the integrated bdist_wheel command (available since setuptools 70.1).
# The venv-bundled version (66.x on Python 3.11) lacks bdist_wheel, causing
# the editable install to fail when the separate 'wheel' package is absent.
#
# To update: change the version, download the new wheel, and replace the hash:
#   pip download setuptools==<new-version> --no-deps -d /tmp/st-dl
#   pip hash /tmp/st-dl/setuptools-<new-version>-py3-none-any.whl
